defmodule TiannaraWeb.GlobalHealthLive do
  use Phoenix.LiveView

  alias Tiannara.Telemetry.Stream
  alias Tiannara.CIS.StateStore

  def mount(_params, _session, socket) do
    :ok = Phoenix.PubSub.subscribe(Tiannara.PubSub, "global:collapse_risk")
    :ok = Phoenix.PubSub.subscribe(Tiannara.PubSub, "cis:alerts")

    {:ok,
     assign(socket,
       global_health: %{},
       alerts: [],
       civilizations: []
     )}
  end

  def handle_info({:cis_alert, alert}, socket) do
    {:noreply,
     update(socket, :alerts, fn alerts ->
       [alert | Enum.take(alerts, 50)]
     end)}
  end

  def handle_info({:global_risk, risk}, socket) do
    {:noreply, assign(socket, :global_health, risk)}
  end
end
