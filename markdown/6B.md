defmodule Tiannara.ALGL do
  @moduledoc """
  Adaptive Law Genesis Layer (ALGL)

  Fully OTS-gated ontology mutation engine.
  """

  alias Tiannara.OTS
  alias Tiannara.MSCL
  alias Tiannara.OCM
  alias Tiannara.CIS

  require Logger

  # ==========================================================
  # PUBLIC API
  # ==========================================================

  def mutate(artifact, context \\ %{}) do
    with {:ok, ots_record} <- OTS.validate_artifact(artifact, context),
         {:ok, mutation} <- propose_mutation(ots_record),
         {:ok, validated_mutation} <- validate_mutation(mutation, ots_record),
         {:ok, committed} <- execute_mutation(validated_mutation, ots_record) do
      {:ok, committed}
    else
      {:error, reason} ->
        Logger.warning("ALGL mutation rejected: #{inspect(reason)}")
        {:error, reason}
    end
  end

  # ==========================================================
  # STEP 1 — PROPOSAL
  # ==========================================================

  defp propose_mutation(ots_record) do
    mutation =
      ots_record.ast
      |> random_transform()
      |> annotate_with_semantics(ots_record)

    {:ok, mutation}
  end

  # ==========================================================
  # STEP 2 — VALIDATION PIPELINE
  # ==========================================================

  defp validate_mutation(mutation, ots_record) do
    with :ok <- OCM.check_equivalence(mutation, ots_record),
         :ok <- MSCL.check_constraints(mutation),
         :ok <- CIS.safety_check(mutation) do
      {:ok, mutation}
    end
  end

  # ==========================================================
  # STEP 3 — EXECUTION
  # ==========================================================

  defp execute_mutation(mutation, ots_record) do
    case Tiannara.MetaCausalityCompiler.validate(mutation) do
      {:ok, compiled} ->
        {:ok, deploy(compiled, ots_record)}

      {:error, reason} ->
        {:error, {:compile_rejection, reason}}
    end
  end

  # ==========================================================
  # INTERNAL OPERATIONS
  # ==========================================================

  defp random_transform(ast) do
    Macro.postwalk(ast, fn node ->
      case :rand.uniform(10) do
        1 -> mutate_operator(node)
        _ -> node
      end
    end)
  end

  defp mutate_operator(node), do: node

  defp annotate_with_semantics(mutation, ots_record) do
    Map.put(mutation, :origin_equivalence, ots_record.equivalence_class_id)
  end

  defp deploy(compiled, _ots_record) do
    Code.compile_quoted(compiled)
  end
end
