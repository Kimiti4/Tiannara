    end
  end

  def store(observer_id, memory) do
    :ets.insert(@chronogram_table, {observer_id, random_phase(), memory})

    #version 310 es

    layout(local_size_x = 16, local_size_y = 16) in;

    layout(rgba32f, binding = 0)

    "float #{d} = #{a} * #{b};"
  end

  defp emit(%Instruction{opcode: :div, dest: d, src_a: a, src_b: b}) do
