        this.gl = glContext;
        this.pipeline = computePipeline;
        this.ssbo = null;

            threshold: gl.getUniformLocation(this.program, "u_discontinuity_threshold")
        };
    }

    compileComputeProgram(source) {
        const gl = this.gl;
        const shader = gl.createShader(gl.COMPUTE_SHADER);
        gl.shaderSource(shader, source);
        gl.compileShader(shader);

        gl.attachShader(prog, shader);
        gl.linkProgram(prog);

        gl.compileShader(shader);

        gl.attachShader(program, shader);
        gl.linkProgram(program);
