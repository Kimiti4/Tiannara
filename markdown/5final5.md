defmodule Tiannara.Topology.PersistentHomology do
  @moduledoc """
  Approximates persistent homology H0/H1 via filtration + cycle basis.
  Returns persistence diagram and topological anchors for DFG folding.
  """

  @type edge :: %{u: any(), v: any(), weight: number()}
  @type persistence_feature :: %{birth: number(), death: number(), persistence: number(), type: :h0 | :h1}

  @doc """
  Computes approximate persistence diagram from weighted graph edges.
  """
  @spec compute([edge()], persistence_threshold :: number()) :: %{diagram: [persistence_feature()], anchors: [persistence_feature()]}
  def compute(edges, threshold \\ 0.5) do
    sorted = Enum.sort_by(edges, & &1.weight, :asc)

    {h0_features, h1_features, _uf_state, _tree} =
      Enum.reduce(sorted, {[], [], init_union_find(sorted), MapSet.new()}, fn edge, acc ->
        add_edge(edge, acc)
      end)

    diagram = Enum.concat(h0_features, h1_features)
    anchors = Enum.filter(diagram, &(&1.persistence > threshold))

    %{diagram: diagram, anchors: anchors}
  end

  defp init_union_find(edges) do
    nodes = Enum.flat_map(edges, &[&1.u, &1.v]) |> Enum.uniq()
    Enum.reduce(nodes, %{}, fn n, map -> Map.put(map, n, %{parent: n, rank: 0}) end)
  end

  defp find(uf, x) do
    case Map.fetch!(uf, x) do
      %{parent: ^x} -> x
      %{parent: p} ->
        root = find(uf, p)
        {root, Map.put(uf, x, %{parent: root, rank: Map.fetch!(uf, x).rank})}
    end
  end

  defp union(uf, x, y) do
    {root_x, uf1} = find(uf, x)
    {root_y, uf2} = find(uf1, y)

    cond do
      root_x == root_y -> {uf2, false}
      true ->
        rank_x = Map.fetch!(uf2, root_x).rank
        rank_y = Map.fetch!(uf2, root_y).rank

        new_uf =
          if rank_x < rank_y do
            Map.update!(uf2, root_x, &%{&1 | parent: root_y})
          else
            Map.update!(uf2, root_y, &%{&1 | parent: root_x})
            |> Map.update!(root_x, fn node ->
              if rank_x == rank_y, do: %{node | rank: node.rank + 1}, else: node
            end)
          end

        {new_uf, true}
    end
  end

  defp add_edge(%{u: u, v: v, weight: w}, {h0, h1, uf, tree}) do
    case union(uf, u, v) do
      {new_uf, true} ->
        # Edge connects components → H0 feature born/death tracked implicitly
        {h0, h1, new_uf, MapSet.put(tree, {u, v})}

      {new_uf, false} ->
        # Edge creates cycle → H1 feature born at `w`, death tracked later
        {h0, [%{birth: w, death: :infinity, persistence: :infinity, type: :h1} | h1], new_uf, tree}
    end
  end
end


### 📐 Mathematical Notes
- **H0 Approximation**: Connected components merge as edges are added. Birth = first node appearance, death = merge weight. (Tracked implicitly; full version adds explicit component tracking.)
- **H1 Approximation**: Edges that close cycles form fundamental cycle basis. Birth = edge weight. Death = when cycle is collapsed or filtered out.
- **Anchors**: Features with persistence > threshold become topological anchors $\alpha$ for DFG manifold embedding.
- **Complexity**: $O(|E|\log|E|)$ due to sorting + near-linear Union-Find.
