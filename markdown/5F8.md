  def detect_resonance(observer_a, observer_b, entropy_series_a, entropy_series_b) do

      cosine_similarity(entropy_series_a, entropy_series_b)

    if correlation > @resonance_threshold do

    else
      :stable_isolation
    end
  end

  defp cosine_similarity(a, b) do
    dot =
      Enum.zip(a, b)
      |> Enum.map(fn {x, y} -> x * y end)
      |> Enum.sum()

    mag_a = :math.sqrt(Enum.sum(Enum.map(a, &(&1 * &1))))
    mag_b = :math.sqrt(Enum.sum(Enum.map(b, &(&1 * &1))))

    dot / ((mag_a * mag_b) + 0.0001)
  end
