defmodule Tiannara.AAL.LexicalTensegrity do
@moduledoc "L9: Deterministic NLG — Template-driven, truth-anchored, tone-modulated"
require Logger

@templates %{
success: "Processed: {{query}}. Result: {{result}}.",
fallback: "I cannot provide a fully confident response at this time. Options: (a) gather context, (b) simulate, (c) escalate.",
anomaly: "Alert: {{type}} detected in {{component}}. Action: {{action}}. Coherence: {{score}}%"
}

@valid_terms MapSet.new(["processed", "result", "alert", "detected", "component", "action", "coherence", "confident", "response", "gather", "simulate", "escalate"])

@spec synthesize(String.t(), map(), list(map())) :: String.t()
def synthesize(query, ctx, anticipations) do
result = format_context(ctx, anticipations)
base = apply_template(:success, %{query: query, result: result})
apply_tone(base, ctx)
end

@spec fallback(atom()) :: String.t()
def fallback(:coherence*rejected), do: @templates.fallback
def fallback(:truth_anchor_violation), do: "Response filtered: unanchored vocabulary detected."
def fallback(*), do: @templates.fallback

@spec synthesize_alert(String.t(), String.t(), String.t(), float()) :: String.t()
def synthesize_alert(type, component, action, coherence) do
score_pct = Float.round(coherence \* 100, 1)
base = apply_template(:anomaly, %{type: type, component: component, action: action, score: "#{score_pct}"})
apply_tone(base, %{urgency: true})
end

defp apply_template(key, params) do
Enum.reduce(params, Map.get(@templates, key, ""), fn {k, v}, acc ->
String.replace(acc, "{{#{k}}}", anchor_term(to_string(v)))
end)
end

defp anchor*term(term) do
words = String.split(term, ~r/\s+/, trim: true)
case Enum.find(words, fn w -> !MapSet.member?(@valid_terms, String.downcase(w)) end) do
nil -> term
* -> "[UNANCHORED]"
end
end

defp format_context(ctx, anticipations) do
anticipations
|> Enum.map(fn a -> "#{a.description} (#{Float.round(a.confidence, 2)})" end)
|> Enum.join(", ")
|> case do
"" -> "Context validated."
res -> "Anticipations: #{res}."
end
end

defp apply*tone(text, %{urgency: true}) do
String.replace(text, ~r/\b(perhaps|might|could)\b/i, "") |> String.trim()
end
defp apply_tone(text, *), do: text
end
