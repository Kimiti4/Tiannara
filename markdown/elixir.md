defmodule TiannaraRuntime.NATS.Subscriber do

  use GenServer

  require Logger



  alias Gnat.ConnectionSupervisor



  def start_link(_opts \\ []) do

    GenServer.start_link(__MODULE__, %{}, name: __MODULE__)

  end



  @impl true

  def init(state) do

    {:ok, gnat} = ConnectionSupervisor.start_link(%{

      connection_settings: [

        %{host: "127.0.0.1", port: 4222}

      ]

    })



    {:ok, _subscription_pid} =

      Gnat.sub(gnat, self(), "tiannara.ecology.state")



    Logger.info("🧠 NATS Subscriber connected")



    {:ok, Map.put(state, :gnat, gnat)}

  end



  @impl true

  def handle_info({:msg, msg}, state) do

    payload =

      msg.body

      |> Jason.decode!()



    Logger.info("📥 Received ecological state")



    IO.inspect(payload)



    evaluate_cis(payload)



    {:noreply, state}

  end



  defp evaluate_cis(state) do

    entropy = state["entropy"]

    dominance = state["dominance"]



    cond do

      dominance > 0.9 ->

        Logger.warning("⚠️ EXTREME DOMINANCE DETECTED")



      entropy < 0.5 ->

        Logger.warning("⚠️ ENTROPY COLLAPSE RISK")



      true ->

        :ok

    end

  end

end
